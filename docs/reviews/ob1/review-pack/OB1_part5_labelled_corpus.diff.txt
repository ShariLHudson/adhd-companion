+export type CorpusContextId =
+  | "open_question" // Spark asked an open question (e.g. an intake question)
+  | "offer_single" // Spark offered one action ("Want me to…?") on the previous turn

+  | "suspended_work"; // earlier work is suspended; an unrelated topic is active
