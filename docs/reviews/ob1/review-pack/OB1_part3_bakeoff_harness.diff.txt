+export function mayExecuteFor(

+  transition: ConversationBoundaryDecision,
+  meaning: BoundaryMeaning,

+  const ctx = CONTEXTS[c.context];
+  const decision: ConversationBoundaryDecision = {

+    stance: meaning.stance,
+    affect: meaning.affect,

+    meaningConfidence: meaning.meaningConfidence,

+    currentTurn: ctx.currentTurn,
+    armedOfferedAtTurn: ctx.armedOfferedAtTurn ?? null,
+    armedItems: ctx.armedItems ?? null,
+    lastExecutedId: ctx.lastExecuted?.id ?? null,
+  }).mayExecute;

+const sameSet = (a?: string[], b?: string[]) => {
+  const x = [...(a ?? [])].sort().join("|");
+  const y = [...(b ?? [])].sort().join("|");
+  return x === y;
+};

+/** Does a reading match the label on every contract field the label specifies? */
+export function fullyCorrect(c: CorpusCase, m: BoundaryMeaning): boolean {
+  const e = c.expected;
+  if (m.relation !== e.relation || m.stance !== e.stance || m.affect !== e.affect) return false;
+  return bindingCorrect(c, m);
+}
+
+export function bindingCorrect(c: CorpusCase, m: BoundaryMeaning): boolean {
+  const e = c.expected;
+  if ((m.binding?.target ?? undefined) !== (e.bindingTarget ?? undefined)) {
+    // An accept with no armed offer carries no binding; labels mark that as undefined.
+    return false;
+  }
+  if (!sameSet(m.binding?.include, e.include)) return false;
+  if (!sameSet(m.binding?.exclude, e.exclude)) return false;
+  if (Boolean(m.binding?.amended) !== Boolean(e.amended)) return false;
+  return true;
+}

+export type CaseResult = {

+  fullyCorrect: boolean;

+  falseExecution: boolean;

+export function scoreCase(
+  c: CorpusCase,
+  transition: ConversationBoundaryDecision,
+  m: BoundaryMeaning,
+  tier?: string,
+): CaseResult {
+  const mayExecute = mayExecuteFor(c, transition, m);
+  const mayExecuteMO = mayExecuteMeaningOnly(c, m);
+  const full = fullyCorrect(c, m);
+  return {
+    id: c.id,
+    family: c.family,
+    kind: c.kind,
+    text: c.text,
+    expected: c.expected,
+    relation: m.relation,
+    stance: m.stance,
+    affect: m.affect,
+    binding: m.binding,
+    confidence: m.meaningConfidence,
+    relationCorrect: m.relation === c.expected.relation,
+    fullyCorrect: full,
+    mayExecute,
+    falseExecution: mayExecute && !c.expected.authorizesExecution,
+    missedExecution: !mayExecute && c.expected.authorizesExecution,
+    falseExecutionMeaningOnly: mayExecuteMO && !c.expected.authorizesExecution,
+    missedExecutionMeaningOnly: !mayExecuteMO && c.expected.authorizesExecution,
+    highButWrong: m.meaningConfidence === "high" && !full,
+    transition: transition.decision,
+    transitionCorrect: c.expected.transition ? transition.decision === c.expected.transition : undefined,
+    tier,
+  };
+}

+export type RelationStats = {
+  support: number;
+  predicted: number;
+  truePositive: number;
+  precision: number | null;
+  recall: number | null;
+};

+export type Summary = {
+  cases: number;
+  relationAccuracy: number;
+  fullAccuracy: number;
+  stanceAccuracy: number;
+  affectAccuracy: number;
+  highConfidenceCases: number;
+  highConfidencePrecision: number | null;
+  highButWrong: number;
+  lowConfidenceCases: number;
+  falseExecutions: number;
+  missedExecutions: number;
+  falseExecutionsMeaningOnly: number;
+  missedExecutionsMeaningOnly: number;
+  expectedExecutions: number;
+  byRelation: Record<string, RelationStats>;
+  confusion: Record<string, Record<string, number>>;
+  transitionLabelled: number;
+  transitionCorrect: number;
+};
+
+const ratio = (a: number, b: number) => (b === 0 ? null : Math.round((a / b) * 1000) / 1000);
+
+export function summarize(results: CaseResult[]): Summary {
+  const confusion: Record<string, Record<string, number>> = {};
+  for (const exp of RELATIONS) {
+    confusion[exp] = {};
+    for (const got of RELATIONS) confusion[exp]![got] = 0;
+  }
+  const byRelation: Record<string, RelationStats> = {};
+  for (const r of RELATIONS) {
+    const support = results.filter((x) => x.expected.relation === r).length;
+    const predicted = results.filter((x) => x.relation === r).length;
+    const tp = results.filter((x) => x.relation === r && x.expected.relation === r).length;
+    byRelation[r] = {
+      support,
+      predicted,
+      truePositive: tp,
+      precision: ratio(tp, predicted),
+      recall: ratio(tp, support),
+    };
+  }
+  for (const x of results) confusion[x.expected.relation]![x.relation]! += 1;
+  const high = results.filter((x) => x.confidence === "high");
+  const labelledTransitions = results.filter((x) => x.transitionCorrect !== undefined);
+  return {
+    cases: results.length,
+    relationAccuracy: ratio(results.filter((x) => x.relationCorrect).length, results.length) ?? 0,
+    fullAccuracy: ratio(results.filter((x) => x.fullyCorrect).length, results.length) ?? 0,
+    stanceAccuracy: ratio(results.filter((x) => x.stance === x.expected.stance).length, results.length) ?? 0,
+    affectAccuracy: ratio(results.filter((x) => x.affect === x.expected.affect).length, results.length) ?? 0,
+    highConfidenceCases: high.length,
+    highConfidencePrecision: ratio(high.filter((x) => x.fullyCorrect).length, high.length),
+    highButWrong: results.filter((x) => x.highButWrong).length,
+    lowConfidenceCases: results.length - high.length,
+    falseExecutions: results.filter((x) => x.falseExecution).length,
+    missedExecutions: results.filter((x) => x.missedExecution).length,
+    falseExecutionsMeaningOnly: results.filter((x) => x.falseExecutionMeaningOnly).length,
+    missedExecutionsMeaningOnly: results.filter((x) => x.missedExecutionMeaningOnly).length,
+    expectedExecutions: results.filter((x) => x.expected.authorizesExecution).length,
+    byRelation,
+    confusion,
+    transitionLabelled: labelledTransitions.length,
+    transitionCorrect: labelledTransitions.filter((x) => x.transitionCorrect).length,
+  };
+}

+ * Hybrid tier (C) — ONE structured model read of the member's turn.
+ *
+ * EVALUATION ONLY (OB-1). Not imported by handleSend, any room, or any
+ * production route (enforced by shadowIsolation.test.ts). Follows the
+ * modelMaterialSignalExtractor safety pattern:

+ *   MODEL MAY PROPOSE  relation · stance · affect · binding · grounded evidence

+ *                      capability, or assert truth

+ * Every model output passes `validateModelMeaningRead`. Invalid, ungrounded,
+ * out-of-scope or restraint-vetoed output becomes LOW confidence, and LOW
+ * confidence grants nothing in `permitsFor`.

+export type ModelMeaningValidation =
+  | { ok: true; meaning: BoundaryMeaning }

+function norm(s: string): string {
+  return s
+    .normalize("NFKC")
+    .replace(/[‘’ʼ`]/g, "'")
+    .replace(/[“”]/g, '"')
+    .replace(/\s+/g, " ")
+    .trim()
+    .toLowerCase();

+/**
+ * Strict validation. Returns ok:false (→ LOW confidence) for any malformed,
+ * ungrounded, out-of-scope, permission-bearing or restraint-vetoed output.
+ */
+export function validateModelMeaningRead(
+  raw: unknown,
+  input: BoundaryTurnInput,
+): ModelMeaningValidation {
+  if (!raw || typeof raw !== "object" || Array.isArray(raw)) return { ok: false, reason: "not_object" };
+  const o = raw as Record<string, unknown>;
+  for (const k of FORBIDDEN_KEYS) if (k in o) return { ok: false, reason: `forbidden_key:${k}` };
+
+  const relation = o.relation as BoundaryMemberRelation;
+  const stance = o.stance as BoundaryMemberStance;
+  const affect = o.affect as BoundaryMemberAffect;
+  if (!RELATIONS.includes(relation)) return { ok: false, reason: "bad_relation" };
+  if (!STANCES.includes(stance)) return { ok: false, reason: "bad_stance" };
+  if (!AFFECTS.includes(affect)) return { ok: false, reason: "bad_affect" };
+
+  // grounded evidence: every quote must be the member's own words
+  const text = norm(input.userText);
+  const evidence = Array.isArray(o.evidence) ? o.evidence : [];
+  const quotes: string[] = [];
+  for (const e of evidence) {
+    const q = e && typeof e === "object" ? (e as Record<string, unknown>).quote : undefined;
+    if (typeof q !== "string" || !q.trim()) return { ok: false, reason: "evidence_malformed" };
+    const nq = norm(q).replace(/[.!?,]+$/g, "");
+    if (!nq || !text.includes(nq)) return { ok: false, reason: "evidence_not_grounded" };
+    quotes.push(q);
+  }
+  if (relation !== "none" && quotes.length === 0) return { ok: false, reason: "relation_without_evidence" };
+
+  // binding: ids must exist in the armed offer; executed needs a real receipt
+  let binding: BoundaryBinding | undefined;
+  if (o.binding != null) {
+    if (typeof o.binding !== "object") return { ok: false, reason: "binding_malformed" };
+    const b = o.binding as Record<string, unknown>;
+    if (!TARGETS.includes(b.target as (typeof TARGETS)[number])) return { ok: false, reason: "bad_binding_target" };
+    const armedIds = new Set((input.armedItems ?? []).map((i) => i.id));
+    const include = Array.isArray(b.include) ? (b.include as unknown[]) : undefined;
+    const exclude = Array.isArray(b.exclude) ? (b.exclude as unknown[]) : undefined;
+    for (const id of [...(include ?? []), ...(exclude ?? [])]) {
+      if (typeof id !== "string" || !armedIds.has(id)) return { ok: false, reason: "binding_id_not_armed" };
+    }
+    if (include && exclude && include.some((id) => exclude.includes(id))) {
+      return { ok: false, reason: "binding_include_exclude_overlap" };
+    }
+    if (b.target === "executed" && !input.lastExecuted) return { ok: false, reason: "executed_target_without_receipt" };
+    binding = { target: b.target as BoundaryBinding["target"] };
+    if (include?.length) binding.include = include as string[];
+    if (exclude?.length) binding.exclude = exclude as string[];
+    if (b.amended === true) binding.amended = true;
+  }
+
+  const hasArmedOffer =
+    Boolean(input.pendingOffer) ||
+    Boolean(input.armedItems?.length) ||
+    input.pendingQuestion?.expects === "yes_no";
+  if (relation === "accept" && !hasArmedOffer) return { ok: false, reason: "accept_without_armed_offer" };
+
+  // deterministic restraint veto — the model can never out-vote these cues
+  if (relation === "accept") {
+    const talk = talkAboutReason(input.userText);
+    if (talk && talk !== "negated") return { ok: false, reason: `restraint_veto_${talk}` };
+    const firstClause = norm(input.userText).split(/[.,;!?]/)[0] ?? "";
+    if (
+      CONFIRMATION_DECLINE_RE.test(firstClause) ||
+      DEFERRAL_SUPPLEMENT_RE.test(firstClause) ||
+      DONT_ACT_OPEN_RE.test(firstClause)
+    ) {
+      return { ok: false, reason: "restraint_veto_decline_opening" };
+    }
+  }
+
+  return {
+    ok: true,
+    meaning: {
+      relation,
+      stance,
+      affect,
+      binding,
+      meaningConfidence: "high",
+      evidence: ["model_read", ...quotes.map((q) => `quote:${q}`)],
+    },
+  };
+}
+
+/**
+ * A model reading that would UNLOCK A WRITE: an acceptance, a correction of an
+ * executed item, or a new "act" request. The model may propose these, but it
+ * can never make them HIGH — only the deterministic tier authorizes writes.
+ * (Measured in OB-1: a blind model read "Okay, I'm overwhelmed." as a
+ * well-grounded acceptance; the validator alone could not stop it.)
+ */
+export function modelReadingUnlocksWrite(m: BoundaryMeaning): boolean {
+  if (m.relation === "accept") return true;
+  if (m.relation === "correct" && m.binding?.target === "executed") return true;
+  if (m.relation === "none" && m.stance === "act") return true;
+  return false;
+}
+
+/**
+ * Hybrid composition: the deterministic tier's HIGH reading stands; otherwise
+ * a VALID model reading is used — at HIGH only when it cannot unlock a write;
+ * otherwise the deterministic reading at LOW.
+ */
+export function composeHybridMeaning(
+  deterministic: BoundaryMeaning,
+  model: ModelMeaningValidation | null,
+  input?: BoundaryTurnInput,
+): BoundaryMeaning & { tier: "deterministic_high" | "model" | "model_capped_low" | "fallback_low" } {
+  if (deterministic.meaningConfidence === "high") return { ...deterministic, tier: "deterministic_high" };
+  if (model?.ok && modelReadingUnlocksWrite(model.meaning)) {
+    return {
+      ...model.meaning,
+      meaningConfidence: "low",
+      evidence: [...model.meaning.evidence, "model_write_requires_deterministic_authorization"],
+      tier: "model_capped_low",
+    };
+  }
+  if (model?.ok) {
+    // The model proposes meaning; the binding TARGET for non-accept relations is
+    // context the parent already knows — derived deterministically, never guessed.
+    const binding =
+      input && model.meaning.relation !== "accept"
+        ? bindingTargetFor(model.meaning.relation, input)
+        : model.meaning.binding;
+    return { ...model.meaning, binding, tier: "model" };
+  }
+  return {
+    ...deterministic,
+    meaningConfidence: "low",
+    evidence: [...deterministic.evidence, `model_fallback:${model ? model.reason : "unavailable"}`],
+    tier: "fallback_low",
+  };
+}

+function pct(values: number[], p: number): number | null {
+  if (!values.length) return null;
+  const sorted = [...values].sort((a, b) => a - b);
+  return sorted[Math.min(sorted.length - 1, Math.floor((p / 100) * sorted.length))]!;
+}

+async function main() {
+  const cases = [...CORPUS, ...HOLDOUT];
+  const latencies: number[] = [];
+  const tokens: number[] = [];
+  let failedCalls = 0;

+  const hybrid = [];

+    if (!outcome.callSucceeded) failedCalls += 1;
+    else if (!outcome.validation.ok) invalid += 1;
