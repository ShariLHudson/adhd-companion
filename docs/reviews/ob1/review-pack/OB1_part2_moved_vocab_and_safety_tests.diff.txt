+/**
+ * OB-1 shadow isolation — proves the meaning contract is computed but NOT
+ * consumed: no production file reads relation / stance / affect / binding /
+ * meaningConfidence, calls permitsFor, or imports the meaning module, the
+ * certification harness, or the evaluation-only model read.
+ */
+import { readdirSync, readFileSync, statSync } from "node:fs";
+import path from "node:path";
+import { describe, expect, it } from "vitest";
+
+const ROOT = path.resolve(__dirname, "../..");
+const SCAN_DIRS = ["app", "components", "lib", "proxy.ts"];
+
+function productionFiles(): string[] {
+  const out: string[] = [];
+  const walk = (p: string) => {
+    const st = statSync(p);
+    if (st.isDirectory()) {
+      if (/node_modules|\.next|__tests__/.test(p)) return;
+      for (const f of readdirSync(p)) walk(path.join(p, f));
+      return;
+    }
+    if (!/\.(ts|tsx)$/.test(p)) return;
+    if (/\.test\.(ts|tsx)$/.test(p)) return;
+    out.push(path.relative(ROOT, p));
+  };
+  for (const d of SCAN_DIRS) {
+    try {
+      walk(path.join(ROOT, d));
+    } catch {
+      // optional entry
+    }
+  }
+  return out;
+}
