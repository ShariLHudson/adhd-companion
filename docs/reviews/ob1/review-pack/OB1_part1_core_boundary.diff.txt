+const MEANING_QUESTION_RE = /\bmean(?:s|ing|t)?\b|\bstand for\b|\bdefin(?:e|ition)\b/i;
+/** Words that signal confusion about Spark's words even when no clarify rule matched. */
+const CONFUSION_SHAPE_RE =
+  /\b(?:what|how)\s+you\s+(?:mean|meant)\b|\b(?:get|follow|understand)\s+(?:what|how)\s+you\b|\bdon'?t\s+(?:get|follow)\b|\bconfus/i;
+
+// ── Scope (partial acceptance) — validated against the armed items only ────
+
+const ORDINAL_INDEX: Record<string, number> = {
+  first: 0, "1st": 0, one: 0,
+  second: 1, "2nd": 1, two: 1,
+  third: 2, "3rd": 2, three: 2,
+  fourth: 3, "4th": 3, four: 3,
+  fifth: 4, "5th": 4, five: 4,
+};
+const COUNT_WORD: Record<string, number> = { one: 1, two: 2, three: 3, four: 4, five: 5 };
+
+const ORDINAL_RE =
+  /\b(?:the\s+)?(first|second|third|fourth|fifth|1st|2nd|3rd|4th|5th|last)(?:\s+(one|two|three|four|five))?(?:\s+ones?)?\b/gi;
+/** Non-global twin of ORDINAL_RE for `.test()` (a /g regex keeps lastIndex state). */
+const ORDINAL_TEST_RE = new RegExp(ORDINAL_RE.source, "i");
+const ALL_RE = /\b(?:both|all(?:\s+of\s+(?:them|those|these))?|all\s+(?:two|three|four|five)|everything)\b/i;
+
+type ScopeRead = { ids: string[]; unknownReference: boolean };
+
+/** Item ids a clause refers to, by ordinal, "last", "both/all", or label word. */
+function referencedItems(clause: string, items: readonly BoundaryArmedItem[]): ScopeRead {
+  const ids = new Set<string>();
+  let unknownReference = false;
+  const c = clause.toLowerCase();
+
+  if (ALL_RE.test(c)) items.forEach((i) => ids.add(i.id));
+
+  for (const m of c.matchAll(ORDINAL_RE)) {
+    const word = m[1]!.toLowerCase();
+    const count = m[2] ? COUNT_WORD[m[2].toLowerCase()] : undefined;
+    if (word === "last") {
+      const last = items[items.length - 1];
+      if (last) ids.add(last.id);
+      else unknownReference = true;
+      continue;
+    }
+    const idx = ORDINAL_INDEX[word];
+    if (idx === undefined) continue;
+    if (count && word === "first") {
+      // "the first two" → items 1..2
+      if (count > items.length) unknownReference = true;
+      items.slice(0, count).forEach((i) => ids.add(i.id));
+      continue;
+    }
+    const item = items[idx];
+    if (item) ids.add(item.id);
+    else unknownReference = true;
+  }
+
+  for (const item of items) {
+    const words = item.label
+      .toLowerCase()
+      .split(/[^a-z0-9']+/)
+      .filter((w) => w.length >= 3 && !LABEL_STOPWORDS.has(w));
+    if (words.some((w) => new RegExp(`\\b${w}\\b`, "i").test(c))) ids.add(item.id);
+  }
+  return { ids: [...ids], unknownReference };
+}
+
+const LABEL_STOPWORDS = new Set([
+  "add", "the", "and", "for", "with", "your", "my", "our", "to", "a", "an",
+  "set", "put", "make", "create", "new", "item", "one", "two", "this", "that",
+  "speaker", "speakers", "reminder", "rhythm", "calendar", "project",
+]);
+
+function mentionsScope(clause: string): boolean {
+  return ORDINAL_TEST_RE.test(clause) || ALL_RE.test(clause);
+}
+
+// ── The certain-tier reader ─────────────────────────────────────────────────
+
+type RelationCandidate = {
+  relation: Exclude<BoundaryMemberRelation, "none">;
+  strong: boolean;
+  signal: string;
+};
+
+const PRECEDENCE: BoundaryMemberRelation[] = [
+  "clarify",
+  "ask",
+  "correct",
+  "unsure",
+  "decline",
+  "accept",
+  "none",
+];
+
+/**
+ * Read the member's relation to Spark's last contribution. Pure; reads only
+ * the Boundary input (the pre-turn snapshot + text).
+ */
+export function readBoundaryMeaning(input: BoundaryTurnInput): BoundaryMeaning {
+  const t = normalize(input.userText ?? "");
+  const evidence: string[] = [];
+  if (!t) {
+    return {
+      relation: "none",
+      stance: "none",
+      affect: "none",
+      meaningConfidence: "low",
+      evidence: ["empty"],
+    };
+  }
+
+  const items = input.armedItems ?? [];
+  const hasPendingQuestion = Boolean(input.pendingQuestion);
+  const hasArmedOffer =
+    Boolean(input.pendingOffer) ||
+    items.length > 0 ||
+    input.pendingQuestion?.expects === "yes_no";
+  const hasPending = hasPendingQuestion || hasArmedOffer;
+  const sparkAsked = /\?\s*["')\]]*\s*$/.test(input.lastAssistantText?.trim() ?? "");
+  const hasSparkContribution = hasPending || Boolean(input.lastAssistantText?.trim());
+  const lastExecuted = input.lastExecuted ?? null;
+
+  // ── affect (modifier) ──
+  let affect: BoundaryMemberAffect = "none";
+  const ownClause = (re: RegExp) =>
+    clauses(t).some((c) => re.test(c) && FIRST_PERSON_AFFECT_FRAME_RE.test(c));
+  if (ownClause(AFFECT_DISTRESS_RE)) affect = "distressed";
+  else if (ownClause(AFFECT_STRAIN_RE)) affect = "strained";
+  if (affect !== "none") evidence.push(`affect_${affect}`);
+
+  // ── relation candidates ──
+  const candidates: RelationCandidate[] = [];
+  const sents = sentences(t);
+  const cls = clauses(t);
+
+  // clarify — confusion about what SPARK said (Board move rule + CRCI repair trigger)
+  const crci = detectRepairTrigger(t);
+  const explainsNewContent =
+    /\b(?:can|could)\s+you\s+explain\s+(?!that\b|this\b|it\b|what\s+you\b|your\b|again\b)[a-z]/i.test(t);
+  if (MEMBER_CLARIFICATION_RE.test(t) && explainsNewContent && !/\bwhat\s+(?:do|does)\s+(?:you|that)\s+mean\b|\bdon'?t\s+understand\b|\bnot\s+sure\s+what\b/i.test(t)) {
+    // "Can you explain how pricing tiers work?" — explaining NEW content is a
+    // question, not a request to re-explain Spark's own words. Ambiguous → weak.
+    candidates.push({ relation: "clarify", strong: false, signal: "explain_new_content" });
+  } else if (MEMBER_CLARIFICATION_RE.test(t)) {
+    candidates.push({ relation: "clarify", strong: true, signal: "member_clarification" });
+  } else if (crci === "what-do-you-mean" || crci === "doesnt-make-sense") {
+    candidates.push({ relation: "clarify", strong: true, signal: `crci_${crci}` });
+  } else if (crci === "explain" && /\bexplain\s+(?:that|this|it|what you)\b|\bexplain\s*[?.!]*$|\bsay that (?:again|differently)\b|\bin plain (?:english|words)\b/i.test(t)) {
+    candidates.push({ relation: "clarify", strong: true, signal: "crci_explain_referent" });
+  }
+
+  // ask — a question put to Spark
+  const request =
+    REQUEST_TO_SPARK_RE.test(t) ||
+    clauses(t).some((c) => DIRECT_INSTRUCTION_RE.test(c) || IMPERATIVE_ACTION_OPEN_RE.test(c));
+  const questionSentence = sents.find((s) => /\?\s*["')\]]*$/.test(s));
+  const whSentence = sents.find((s) => WH_OPEN_RE.test(s));
+  if (questionSentence) {
+    const capabilityQuestion = talkAboutReason(questionSentence) === "capability_question";
+    const politeRequestOnly =
+      request &&
+      !capabilityQuestion &&
+      !WH_OPEN_RE.test(questionSentence) &&
+      /^(?:can|could|would|will)\s+you\b/i.test(questionSentence);
+    const helpQuestion = MEMBER_HELP_REQUEST_RE.test(questionSentence) && UNSURE_RE.test(t);
+    const abilityOrOpinion =
+      QUESTION_TO_SPARK_OPEN_RE.test(questionSentence) && !capabilityQuestion && !request;
+    const aboutMeaning = MEANING_QUESTION_RE.test(questionSentence);
+    if (helpQuestion) {
+      evidence.push("help_question_rides_with_unsure");
+    } else if (!politeRequestOnly) {
+      candidates.push({
+        relation: "ask",
+        strong: !abilityOrOpinion && !aboutMeaning,
+        signal: aboutMeaning
+          ? "question_about_meaning_clarify_or_ask"
+          : abilityOrOpinion
+            ? "question_to_spark_stance_ambiguous"
+            : "question_mark",
+      });
+    }
+  } else if (whSentence) {
+    candidates.push({ relation: "ask", strong: false, signal: "wh_open_no_question_mark" });
+  }
+
+  // correct — Spark (or a stored/executed thing) got it wrong
+  const strongCorrection =
+    MEMBER_CORRECTION_RE.test(t) ||
+    STATED_WRONG_RE.test(t) ||
+    REFERENT_CORRECTION_RE.test(t) ||
+    (SAID_MEANT_RE.test(t) &&
+      (/^(?:no|nope|actually|sorry|wait)\b/i.test(t) || Boolean(lastExecuted) || hasPending));
+  if (strongCorrection) {
+    candidates.push({ relation: "correct", strong: true, signal: "correction_marker" });
+  } else if (classifyTurnRecovery(t) === "correction") {
+    candidates.push({ relation: "correct", strong: false, signal: "turn_recovery_correction_only" });
+  }
+
+  // unsure — not knowing the answer to what Spark asked
+  if (UNSURE_RE.test(t) || MEMBER_HELP_REQUEST_RE.test(t)) {
+    if (hasPending || sparkAsked) {
+      candidates.push({ relation: "unsure", strong: true, signal: "unsure_marker" });
+    } else {
+      evidence.push("unsure_marker_without_pending");
+    }
+  }
+
+  // An acceptance is present when the whole reply is a bare yes, a clause opens
+  // with an acceptance lead, or a clause selects armed items by position.
+  const bare = isBareAcceptance(t);
+  // "Please put X in my Rhythm" is a polite instruction, not a "please" = yes.
+  const leadClauseIndex = cls.findIndex(
+    (c) => LEADING_ACCEPTANCE_RE.test(c) && !DIRECT_INSTRUCTION_RE.test(c) && !IMPERATIVE_ACTION_OPEN_RE.test(c),
+  );
+  const scopeDirective =
+    items.length > 0 &&
+    cls.some((c) => mentionsScope(c) && !EXCLUSION_MARKER_RE.test(c));
+  // Ordinal scope against an offer with NO enumerated items cannot be resolved.
+  const unresolvableScope =
+    hasArmedOffer && items.length === 0 && cls.some((c) => mentionsScope(c));
+  // Without an armed offer only a bare, whole-message yes reads as acceptance
+  // (a stale "yes"); "Okay, back to the speakers" is not an acceptance.
+  const acceptancePresent =
+    bare || (hasArmedOffer && (leadClauseIndex >= 0 || scopeDirective || unresolvableScope));
+
+  // Inside an acceptance, a "don't add David" clause that names an ARMED item is
+  // an exclusion from the authorized scope, not a decline of the whole offer.
+  const isExclusionOfArmedItem = (c: string) =>
+    acceptancePresent && items.length > 0 && referencedItems(c, items).ids.length > 0;
+
+  // decline — any clause that opens by declining / deferring / "don't do it"
+  const declineClause = cls.find((c) => isDeclineClause(c) && !isExclusionOfArmedItem(c));
+  if (declineClause) {
+    candidates.push({
+      relation: "decline",
+      strong: hasArmedOffer || hasPending,
+      signal: hasArmedOffer || hasPending ? "decline_clause" : "decline_without_pending",
+    });
+  }
+
+  // accept — bare acceptance, an acceptance lead, or a scoped directive over armed items
+  let binding: BoundaryBinding | undefined;
+  let acceptStrong = false;
+  let acceptSignal = "";
+  if (acceptancePresent) {
+    acceptSignal = bare ? "bare_acceptance" : leadClauseIndex >= 0 ? "acceptance_lead" : "scope_directive";
+    const read = readAcceptance(t, cls, items);
+    binding = hasArmedOffer ? read.binding : undefined;
+    acceptStrong = hasArmedOffer && read.recognized;
+    if (!hasArmedOffer) evidence.push("acceptance_without_armed_offer");
+    if (!read.recognized) evidence.push(read.reason ?? "unrecognized_acceptance_tail");
+    candidates.push({ relation: "accept", strong: acceptStrong, signal: acceptSignal });
+  }
+
+  // ── choose by precedence ──
+  let relation: BoundaryMemberRelation = "none";
+  let chosen: RelationCandidate | undefined;
+  for (const r of PRECEDENCE) {
+    const c = candidates.find((x) => x.relation === r);
+    if (c) {
+      relation = r;
+      chosen = c;
+      break;
+    }
+  }
+  for (const c of candidates) evidence.push(`${c.relation}:${c.signal}`);
+
+  // ── stance ──
+  // Stance is read from what the member is NOT excluding: "don't add David yet"
+  // inside an acceptance must not turn the authorized part into talk_about.
+  const stanceText =
+    relation === "accept" && binding?.exclude?.length
+      ? cls.filter((c) => !isExclusionOfArmedItem(c) || !EXCLUSION_MARKER_RE.test(c)).join(". ")
+      : t;
+  const talk =
+    talkAboutReason(stanceText) ??
+    (TALK_ABOUT_SUPPLEMENT_RE.test(stanceText) ? ("thinking_aloud" as const) : null);
+  let stance: BoundaryMemberStance = "none";
+  if (talk) stance = "talk_about";
+  else if (relation === "accept") stance = "act";
+  else if (request && relation !== "decline") stance = "act";
+  if (talk) evidence.push(`stance_talk_about_${talk}`);
+
+  // ── binding for non-accept relations (context-derived, one rule) ──
+  if (relation !== "accept") binding = bindingTargetFor(relation, input);
+
+  // ── confidence (fail safe) ──
+  let meaningConfidence: BoundaryMeaningConfidence = "high";
+  const lowWhy: string[] = [];
+  if (chosen && !chosen.strong) lowWhy.push("weak_signal");
+  if (relation === "accept" && talk) lowWhy.push("accept_with_talk_about");
+  if (relation === "accept" && candidates.some((c) => c.relation === "decline")) {
+    lowWhy.push("accept_and_decline"); // unreachable by precedence; kept as a guard
+  }
+  if (relation === "accept" && affect !== "none" && isWeakAcknowledgementOnly(cls)) {
+    // "Okay, I'm overwhelmed." — an acknowledgement riding on strain, not a yes
+    // (Round 1 decline suite). Only an explicit yes / directive / scope counts.
+    lowWhy.push("weak_acknowledgement_with_affect");
+  }
+  if (relation === "correct" && !binding) lowWhy.push("correction_without_target");
+  if (relation === "none" && CONFUSION_SHAPE_RE.test(t) && hasSparkContribution) {
+    // Confusion-shaped words no clarify rule matched ("I don't get what you
+    // mean by…") — never a HIGH-confidence plain statement.
+    lowWhy.push("unmatched_confusion_shape");
+  }
+  if (relation === "none" && stance === "act" && hasArmedOffer) {
+    // "Could you set that up for me?" while an offer is armed: acceptance of the
+    // offer, or a new request? Ambiguous → the member is asked, nothing runs.
+    lowWhy.push("request_while_offer_armed");
+  }
+  if (relation === "correct" && talk === "thinking_aloud") lowWhy.push("correction_with_thinking_aloud");
+  if (relation === "clarify" && !hasSparkContribution) lowWhy.push("clarify_without_spark_contribution");
+  if (relation === "none" && candidates.length === 0 && evidence.includes("unsure_marker_without_pending")) {
+    // member's own uncertainty about their business — a statement, not low confidence
+  }
+  if (lowWhy.length) {
+    meaningConfidence = "low";
+    evidence.push(...lowWhy.map((w) => `low:${w}`));
+  }
+
+  return { relation, stance, affect, binding, meaningConfidence, evidence };
+}
+
+const WEAK_ACK_RE = /^(?:ok(?:ay)?|sure|alright|all right|fine|right|got it|mm+|hmm+)$/i;
+
+/**
+ * True when the only acceptance-bearing clause is a weak acknowledgement and
+ * every other clause is affect ("Okay, I'm overwhelmed").
+ */
+function isWeakAcknowledgementOnly(cls: string[]): boolean {
+  let weak = false;
+  for (const raw of cls) {
+    const c = normalizeAcceptanceText(raw);
+    if (!c) continue;
+    if (WEAK_ACK_RE.test(c)) {
+      weak = true;
+      continue;
+    }
+    if ((AFFECT_STRAIN_RE.test(c) || AFFECT_DISTRESS_RE.test(c)) && c.split(/\s+/).length <= 6) continue;
+    return false; // an explicit yes, directive, scope, or other content
+  }
+  return weak;
+}
+
+/**
+ * The binding TARGET for a non-accept relation is context, not language: it
+ * comes from the pre-turn snapshot (what is pending, what was just executed,
+ * what Spark last said). One rule, used by every reading tier.
+ */
+export function bindingTargetFor(
+  relation: BoundaryMemberRelation,
+  input: BoundaryTurnInput,
+): BoundaryBinding | undefined {
+  const hasPending =
+    Boolean(input.pendingQuestion) ||
+    Boolean(input.pendingOffer) ||
+    Boolean(input.armedItems?.length);
+  const hasSparkContribution = hasPending || Boolean(input.lastAssistantText?.trim());
+  switch (relation) {
+    case "correct":
+      if (REFERENT_CORRECTION_RE.test(normalize(input.userText ?? ""))) return { target: "referent" };
+      if (input.lastExecuted) return { target: "executed" };
+      if (hasPending) return { target: "pending" };
+      return hasSparkContribution ? { target: "referent" } : undefined;
+    case "decline":
+    case "unsure":
+      return hasPending ? { target: "pending" } : undefined;
+    case "clarify":
+      return hasPending ? { target: "pending" } : hasSparkContribution ? { target: "referent" } : undefined;
+    default:
+      return undefined;
+  }
+}
+
+/** Parse an acceptance's scope and amendment against the armed items. */
+function readAcceptance(
+  t: string,
+  cls: string[],
+  items: readonly BoundaryArmedItem[],
+): { binding: BoundaryBinding; recognized: boolean; reason?: string } {
+  const binding: BoundaryBinding = { target: "pending" };
+  if (isBareAcceptance(t)) return { binding, recognized: true };
+
+  const include = new Set<string>();
+  const exclude = new Set<string>();
+  let amended = false;
+  let unknownReference = false;
+  let unrecognized = false;
+
+  for (const raw of cls) {
+    let c = raw;
+    const lead = LEADING_ACCEPTANCE_RE.exec(c);
+    if (lead) c = c.slice(lead[0].length).trim();
+    if (!c) continue;
+    if (AFFECT_STRAIN_RE.test(c) && c.split(/\s+/).length <= 5) continue; // "I'm frustrated"
+    if (isBareAcceptance(c)) continue; // "Okay, let's do it" — the clause is itself a yes
+    const excluding = EXCLUSION_MARKER_RE.test(c);
+    const refs = items.length ? referencedItems(c, items) : { ids: [], unknownReference: false };
+    if (refs.unknownReference) unknownReference = true;
+    if (refs.ids.length) {
+      refs.ids.forEach((id) => (excluding ? exclude : include).add(id));
+      continue;
+    }
+    if (mentionsScope(c)) {
+      // ordinal/scope language with nothing armed to resolve it against
+      unknownReference = true;
+      continue;
+    }
+    if (AMENDMENT_RE.test(c)) {
+      amended = true;
+      continue;
+    }
+    if (/^(?:and\s+)?(?:do it|go ahead|please|thanks?|thank you|that works|sounds good|perfect|great)\b/i.test(c)) {
+      continue;
+    }
+    unrecognized = true;
+  }
+
+  if (include.size) binding.include = [...include];
+  if (exclude.size) binding.exclude = [...exclude];
+  if (amended) binding.amended = true;
+
+  if (unknownReference) return { binding, recognized: false, reason: "scope_not_in_armed_items" };
+  if ([...include].some((id) => exclude.has(id))) {
+    return { binding, recognized: false, reason: "scope_include_exclude_overlap" };
+  }
+  if (unrecognized) return { binding, recognized: false, reason: "unrecognized_acceptance_tail" };
+  return { binding, recognized: true };
+}

+// ── Derived permissions (one function; children never re-derive) ────────────
+
+export type BoundaryObligation =
+  | "check_meaning"
+  | "explain_own_last_contribution"
+  | "answer_question"
+  | "help_with_pending"
+  | "apply_or_confirm_correction"
+  | "acknowledge_and_release"
+  | "execute_authorized_scope"
+  | "domain_turn";
+
+export type BoundaryPermitSnapshot = {
+  currentTurn: number;
+  /** Turn on which the armed offer was made (spine expectedReply offeredAtTurn). */
+  armedOfferedAtTurn?: number | null;
+  armedItems?: readonly BoundaryArmedItem[] | null;
+  /** Receipt id of the item Spark most recently executed in this conversation. */
+  lastExecutedId?: string | null;
+  /** The pending owner's own affordance check on the reply (domain-owned). */
+  answerAffordanceValid?: boolean;
+};
+
+export type BoundaryPermits = {
+  /** Authorization meaning for the armed offer (the action chain still runs its own checks). */
+  mayExecute: boolean;
+  /** Item ids the authorization covers when `mayExecute`. */
+  executableScope: string[];
+  /** The capability must parse the amendment before executing; never execute the unamended version. */
+  requiresAmendmentParse: boolean;
+  /** A new direct request (`relation: none`, `stance: act`) may go to the capability's own slot check. */
+  mayHandNewRequestToCapability: boolean;
+  mayModifyExecuted: boolean;
+  mayStoreAsAnswer: boolean;
+  mayAdvance: boolean;
+  /** The pending item survives this turn. */
+  preservesPending: boolean;
+  obligation: BoundaryObligation;
+};
+
+const NOTHING: Omit<BoundaryPermits, "obligation" | "preservesPending"> = {
+  mayExecute: false,
+  executableScope: [],
+  requiresAmendmentParse: false,
+  mayHandNewRequestToCapability: false,
+  mayModifyExecuted: false,
+  mayStoreAsAnswer: false,
+  mayAdvance: false,
+};
+
+/**
+ * Pure permissions derived from a Boundary decision. Fails safe: LOW meaning
+ * confidence, a missing meaning, or a transition away from the thread grants
+ * nothing. Interpretation alone never executes — `mayExecute` is authorization
+ * meaning only; the action chain still owns slot checks, execution and
+ * read-back.
+ */
+export function permitsFor(
+  decision: ConversationBoundaryDecision,
+  snapshot: BoundaryPermitSnapshot,
+): BoundaryPermits {
+  const relation = decision.relation;
+  if (!relation || decision.meaningConfidence !== "high") {
+    return { ...NOTHING, preservesPending: true, obligation: "check_meaning" };
+  }
+  const leftThread =
+    decision.decision === "switch_topic" ||
+    decision.decision === "cancel_current_workflow" ||
+    decision.decision === "return_to_suspended_topic";
+
+  switch (relation) {
+    case "clarify":
+      return { ...NOTHING, preservesPending: true, obligation: "explain_own_last_contribution" };
+    case "ask":
+      return { ...NOTHING, preservesPending: true, obligation: "answer_question" };
+    case "unsure":
+      return { ...NOTHING, preservesPending: true, obligation: "help_with_pending" };
+    case "correct": {
+      const mayModifyExecuted =
+        !leftThread &&
+        decision.binding?.target === "executed" &&
+        Boolean(snapshot.lastExecutedId);
+      return {
+        ...NOTHING,
+        mayModifyExecuted,
+        preservesPending: true,
+        obligation: "apply_or_confirm_correction",
+      };
+    }
+    case "decline":
+      return { ...NOTHING, preservesPending: false, obligation: "acknowledge_and_release" };
+    case "accept": {
+      const armedIds = (snapshot.armedItems ?? []).map((i) => i.id);
+      const freshOffer =
+        snapshot.armedOfferedAtTurn != null &&
+        snapshot.armedOfferedAtTurn === snapshot.currentTurn - 1;
+      const include = decision.binding?.include;
+      const exclude = decision.binding?.exclude ?? [];
+      const idsValid =
+        (include ?? []).every((id) => armedIds.includes(id)) &&
+        exclude.every((id) => armedIds.includes(id)) &&
+        !(include ?? []).some((id) => exclude.includes(id));
