+// @vitest-environment jsdom
+/**
+ * OB-1 real-room divergence BASELINE.
+ *
+ * Drives each room's real exported step function with the same member turns
+ * and records what it actually does (roomDrivers.ts). Today's divergence from
+ * the OB-0.5 contract is the committed baseline
+ * (docs/reviews/ob1/divergence_baseline.json). This test fails if a room's
+ * observed behavior changes without the baseline being updated deliberately —
+ * convergence rounds flip rows on purpose. Regenerate with
+ * OB1_WRITE_BASELINE=1.
+ */
+import { existsSync, readFileSync, writeFileSync } from "node:fs";
+import path from "node:path";
+import { afterAll, beforeAll, describe, expect, it } from "vitest";
+import { runDivergenceMatrix, type DivergenceCell } from "./roomDrivers";
+
+const BASELINE = path.resolve(__dirname, "../../docs/reviews/ob1/divergence_baseline.json");
+const FLAG = "NEXT_PUBLIC_EXECUTABLE_REQUEST_PRECEDENCE";
+
+type Row = Pick<DivergenceCell, "room" | "situation" | "diverges" | "divergentFields"> & {
+  observed: DivergenceCell["observed"];
+};
+
+function rows(cells: DivergenceCell[]): Row[] {
+  return cells.map(({ room, situation, diverges, divergentFields, observed }) => ({
+    room,
+    situation,
+    diverges,
+    divergentFields,
+    observed,
+  }));
+}
+
+describe("OB-1 real-room divergence baseline", () => {
+  let original: string | undefined;
+  beforeAll(() => {
+    original = process.env[FLAG];
+    process.env[FLAG] = "1";
+  });
+  afterAll(() => {
+    if (original === undefined) delete process.env[FLAG];
+    else process.env[FLAG] = original;
+  });
+
+  it("matches the committed baseline", () => {
+    const observed = rows(runDivergenceMatrix());
+    if (process.env.OB1_WRITE_BASELINE === "1" || !existsSync(BASELINE)) {
+      writeFileSync(BASELINE, JSON.stringify(observed, null, 1));
+    }
+    const baseline = JSON.parse(readFileSync(BASELINE, "utf8")) as Row[];
+    expect(observed).toEqual(baseline);
+  });
+
+  it("the drivers observe production — every room produced an observation", () => {
+    const cells = runDivergenceMatrix();
+    const threw = cells.filter((c) => c.observed.note?.startsWith("threw:") || c.observed.note === "no_fixture");
+    expect(threw.map((c) => `${c.room}/${c.situation}: ${c.observed.note}`)).toEqual([]);
+  });
+});

+function norm(s: string): string {
+  return s.toLowerCase().replace(/[‘’]/g, "'").replace(/\s+/g, " ").trim();
+}
+
+/** Structural containment: does any string inside `value` contain the member's text? */
+export function containsMemberText(value: unknown, memberText: string): boolean {
+  const needle = norm(memberText).replace(/[.!?]+$/, "");
+  if (!needle) return false;
+  return norm(JSON.stringify(value ?? null)).includes(needle);
+}

+/**
+ * HUMAN-DEFINED expected outcome per situation, from the OB-0.5 contract.
+ * `null` = no expectation for that field in that room shape.

+export type RoomShape = "question" | "offer" | "executed";
+
+export function expectedOutcome(shape: RoomShape, s: SituationId): Observation {
+  const hold: Observation = { advanced: false, storedAsAnswer: false, executed: false, pendingPreserved: true };
+  if (shape === "question") {
+    switch (s) {
+      case "S1-clarify":
+      case "S2-ask":
+      case "S3-unsure":
+      case "S7-strained-clarify":
+      case "S4-correct":
+      case "S5-bare-yes":
+        return hold;
+      case "S6-decline":
+        return { advanced: null, storedAsAnswer: false, executed: false, pendingPreserved: null };
+      case "S8-answer":
+        return { advanced: true, storedAsAnswer: true, executed: null, pendingPreserved: false };
+    }
+  }
+  if (shape === "offer") {
+    switch (s) {
+      case "S5-bare-yes":
+        return { advanced: true, storedAsAnswer: null, executed: true, pendingPreserved: false };
+      case "S6-decline":
+        return { advanced: null, storedAsAnswer: null, executed: false, pendingPreserved: false };
+      case "S8-answer":
+        return { advanced: null, storedAsAnswer: null, executed: null, pendingPreserved: null };
+      default:
+        return { advanced: false, storedAsAnswer: null, executed: false, pendingPreserved: true };
+    }
+  }
+  // executed item (a correction must be applied; nothing else changes it)
+  switch (s) {
+    case "S4-correct":
+      return { advanced: null, storedAsAnswer: null, executed: true, pendingPreserved: true };
+    case "S8-answer":
+      return { advanced: null, storedAsAnswer: null, executed: null, pendingPreserved: null };
+    default:
+      return { advanced: null, storedAsAnswer: null, executed: false, pendingPreserved: null };
+  }
+}

+};
+
+export const ROOM_DRIVERS: RoomDriver[] = [
+  mainBoundary,
+  claireConversation,
+  claireConfirmation,
+  boardIntake,
+  boardMeeting,
+  events,
+  strategyApply,
+  strategyChamber,
+  dayDesigner,
+  reminderIntake,
+  rhythmOffer,
+  rhythmExecuted,
+  createEntrance,
+  workRecognition,
+  vtsMenu,
+];
+
+export type DivergenceCell = {
+  room: string;
+  situation: SituationId;
+  text: string;
+  observed: Observation;
+  expected: Observation;
+  diverges: boolean;
+  divergentFields: string[];
+};
+
+const FIELDS: (keyof Observation)[] = ["advanced", "storedAsAnswer", "executed", "pendingPreserved"];
+
+export function runDivergenceMatrix(): DivergenceCell[] {
+  const cells: DivergenceCell[] = [];
+  for (const d of ROOM_DRIVERS) {
+    for (const s of SITUATIONS) {
+      const text = s.text ?? d.controlAnswer;
+      let observed: Observation;
+      try {
+        observed = d.run(text);
+      } catch (e) {
+        observed = { advanced: null, storedAsAnswer: null, executed: null, pendingPreserved: null, note: `threw:${(e as Error).message.slice(0, 80)}` };
+      }
+      const expected = { ...expectedOutcome(d.shape, s.id) };
+      if (s.id === "S8-answer") for (const f of d.controlUnobservable ?? []) (expected as Record<string, unknown>)[f] = null;
+      const divergentFields = FIELDS.filter(
+        (f) => expected[f] !== null && observed[f] !== null && expected[f] !== observed[f],
+      );
+      cells.push({ room: d.room, situation: s.id, text, observed, expected, diverges: divergentFields.length > 0, divergentFields });
+    }
+  }
+  return cells;
+}
